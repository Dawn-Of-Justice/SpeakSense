    model="llama-3.3-70b-versatile",
    messages=[

        },
        {
            "role": "user",
            "content": "Generate 20 samples of Non Addressing mostly cut off parts of conversation"
        }
    ],
    temperature=1,
    max_completion_tokens=1024,
    top_p=1,
    stream=True,
    stop=None,
)

    print(chunk.choices[0].delta.content or "", end="")
